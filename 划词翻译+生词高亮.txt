    function matchesDomainPattern(host, pat) {
        if (!host || !pat) return false;
        let cleanHost = String(host).toLowerCase().trim();
        let cleanPat = String(pat).toLowerCase().trim();
        cleanPat = cleanPat.replace(/^[a-zA-Z]+:\/\//, '').split('/')[0].split(':')[0].trim();
        if (!cleanPat) return false;
        if (cleanPat.startsWith('*.')) {
            const root = cleanPat.slice(2);
            return cleanHost === root || cleanHost.endsWith('.' + root);
        }
        return cleanHost === cleanPat || cleanHost.endsWith('.' + cleanPat);
    }

            return list.some(pat => matchesDomainPattern(host, pat));
        } catch (e) {

    // ================= 基础工具函数 =================
    function normalizeWord(word) {
        return String(word || '').toLowerCase().trim();
    }

    function escapeHtml(str) {
        return String(str).replace(/[&<>"']/g, s => ({
            '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;'
        }[s]));
    }

    const PRESERVED_BLOCK_SELECTOR = 'h1,h2,h3,h4,h5,h6,li,p,blockquote,figcaption,td,th,dt,dd,pre,tr,figure,summary,details';
    const GENERIC_BLOCK_TAGS = new Set([
        'DIV', 'SECTION', 'ARTICLE', 'MAIN', 'ASIDE', 'HEADER', 'FOOTER', 'NAV'
    ]);

    function normalizeBlockText(text) {
        return String(text || '').replace(/\s+/g, ' ').trim();
    }

    function isLikelyChineseText(text) {
        const value = String(text || '');
        const hanCount = (value.match(/[\u3400-\u9FFF\uF900-\uFAFF]/g) || []).length;
        if (!hanCount) return false;

        const latinCount = (value.match(/[A-Za-z]/g) || []).length;
        const signalCount = hanCount + latinCount;
        return latinCount === 0 || hanCount / signalCount >= 0.35;
    }

    function getSelectedTextBlock(textNode) {
        let element = textNode.parentElement;
        let nearestGenericBlock = null;

        while (element) {
            // 语义块优先于内层通用 DIV。例如 <li><div>...</div></li> 仍应保留为列表项。
            if (element.matches(PRESERVED_BLOCK_SELECTOR)) return element;
            if (!nearestGenericBlock && GENERIC_BLOCK_TAGS.has(element.tagName)) {
                nearestGenericBlock = element;
            }
            element = element.parentElement;
        }

        return nearestGenericBlock;
    }

    function getSelectedTextBlockType(element) {
        if (!element) return 'line';

        const tagName = element.tagName.toLowerCase();
        // 新闻网站（包括 Reuters）常用 DIV 承载正文段落，统一按段落渲染。
        if (GENERIC_BLOCK_TAGS.has(element.tagName)) return 'p';
        return tagName;
    }

    function extractSelectedBlocks(fragment) {
        const blocks = [];
        let currentBlock = null;

        function flushCurrentBlock() {
            if (!currentBlock) return;

            const text = normalizeBlockText(currentBlock.rawText);
            if (text) {
                blocks.push({
                    type: currentBlock.type,
                    text
                });
            }
            currentBlock = null;
        }

        const walker = document.createTreeWalker(
            fragment,
            NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT
        );

        while (walker.nextNode()) {
            const node = walker.currentNode;

            if (node.nodeType === Node.ELEMENT_NODE && node.tagName === 'BR') {
                flushCurrentBlock();
                continue;
            }
            if (node.nodeType !== Node.TEXT_NODE) continue;

            const ownerBlock = getSelectedTextBlock(node);
            const blockKey = ownerBlock || fragment;
            const blockType = getSelectedTextBlockType(ownerBlock);

            if (!currentBlock || currentBlock.key !== blockKey) {
                flushCurrentBlock();
                currentBlock = {
                    key: blockKey,
                    type: blockType,
                    rawText: ''
                };
            }
            currentBlock.rawText += node.nodeValue || '';
        }

        flushCurrentBlock();
        return blocks;
    }

    function getCompleteSelectionBlocks(rawText, detectedBlocks) {
        const canonicalize = value => String(value || '').replace(/\s+/g, '');
        const detectedText = detectedBlocks.map(block => block.text).join('');

        if (detectedBlocks.length && canonicalize(detectedText) === canonicalize(rawText)) {
            return detectedBlocks;
        }

        // 格式识别不完整时只降级结构，绝不能丢弃 selection.toString() 中的任何正文。
        const lines = rawText.split(/\r?\n+/).map(normalizeBlockText).filter(Boolean);
        if (lines.length > 1) {
            return lines.map(line => ({ type: 'line', text: line }));
        }
        return [{ type: 'text', text: rawText }];
    }

    function getSelectionFormat(selection) {
        const rawText = selection.toString().trim();
        if (!rawText || !selection.rangeCount) return null;

        const fragment = selection.getRangeAt(0).cloneContents();
        const blocks = getCompleteSelectionBlocks(rawText, extractSelectedBlocks(fragment));
        const text = (blocks && blocks.length > 0)
            ? blocks.map(block => block.text).join('\n')
            : rawText;

        return {
            text,
            blocks
        };
    }

    function playAudio(text) {
        if (!text) return;
        stopCurrentAudio();

        const engine = CONFIG.ttsEngine || 'youdao';

        // 0. 如果配置为本地原生伴侣服务 (Sherpa-ONNX / Kokoro C++ 极速引擎)
        if (engine === 'local_service') {
            playLocalServiceTTS(text);
            return;
        }

        // 1. 如果配置为 Kokoro 浏览器离线神经网络引擎
        if (engine === 'kokoro') {
            playKokoroTTS(text);
            return;
        }

        // 2. 如果配置为 Edge 自然发音（或在 edge 选项下）
        if (engine === 'edge') {
            if (isEdgeBrowser() && window.speechSynthesis) {
                playEdgeJennyTTS(text);
                return;
            }
            // 非 Edge 环境下自动回退到有道
            playYoudaoTTS(text);
            return;
        }

        // 3. 默认有道发音
        playYoudaoTTS(text);
    }

    function playLocalServiceTTS(text) {
        const baseUrl = (CONFIG.localTtsUrl || GM_getValue('lexipro_local_tts_url', 'http://127.0.0.1:18080')).replace(/\/+$/, '');
        const voice = CONFIG.kokoroVoice || 'af_heart';
        const url = `${baseUrl}/tts?text=${encodeURIComponent(text)}&voice=${encodeURIComponent(voice)}&speed=1.0`;

        stopCurrentAudio();
        const audio = new Audio(url);
        currentAudio = audio;

        const playPromise = audio.play();
        if (playPromise && typeof playPromise.catch === 'function') {
            playPromise.catch(err => {
                if (err && (err.name === 'NotAllowedError' || err.name === 'AbortError')) {
                    return;
                }
                console.warn('[LexiPro TTS] 本地原生伴侣服务未响应 (请运行 start-local-tts.bat)，平滑降级至在线语音:', err);
                // 优雅降级至有道发音
                playYoudaoTTS(text);
            });
        }
    }

    function playKokoroTTS(text) {
        if (typeof chrome !== 'undefined' && chrome.runtime && typeof chrome.runtime.sendMessage === 'function') {
            try {
                chrome.runtime.sendMessage({
                    type: 'KOKORO_TTS_PLAY',
                    text: text,
                    voice: CONFIG.kokoroVoice || 'af_heart',
                    speed: 1.0
                }, (response) => {
                    if (chrome.runtime.lastError || !response || response.success === false) {
                        console.warn('[LexiPro TTS] Kokoro 播放失败，平滑降级至有道/系统语音:', chrome.runtime.lastError?.message || response?.error);
                        playYoudaoTTS(text);
                    }
                });
                return;
            } catch (e) {
                console.warn('[LexiPro TTS] 发送 Kokoro 指令异常，回退有道:', e);
                playYoudaoTTS(text);
                return;
            }
        }
        // 油猴脚本或无扩展宿主环境：回退到有道词典
        playYoudaoTTS(text);
    }

    function playYoudaoTTS(text) {
        try {
            const url = `https://dict.youdao.com/dictvoice?audio=${encodeURIComponent(text)}&type=2`;
            currentAudio = new Audio(url);
            const playPromise = currentAudio.play();
            if (playPromise && typeof playPromise.catch === 'function') {
                playPromise.catch(err => {
                    // 若受浏览器用户交互限制 (NotAllowedError) 或用户快速切换中断 (AbortError)，优雅退出且绝不在控制台打 console.error
                    if (err && (err.name === 'NotAllowedError' || err.name === 'AbortError')) {
                        return;
                    }
                    fallbackSpeechSynthesis(text);
                });
            }
        } catch (e) {
            fallbackSpeechSynthesis(text);
        }
    }

    function playEdgeJennyTTS(text) {
        const utterance = new SpeechSynthesisUtterance(text);
        const voices = window.speechSynthesis.getVoices();
        let selectedVoice = voices.find(v => v.name === 'Microsoft Jenny Online (Natural) - English (United States)') ||
                            voices.find(v => v.name.includes('Jenny') && v.name.includes('Online')) ||
                            voices.find(v => v.name.includes('Online') && v.lang.startsWith('en'));

        if (selectedVoice) {
            utterance.voice = selectedVoice;
            utterance.lang = selectedVoice.lang;
        } else {
            playYoudaoTTS(text);
            return;
        }
        utterance.rate = 0.95;
        window.speechSynthesis.speak(utterance);
    }

    function fallbackSpeechSynthesis(text) {
        if (!window.speechSynthesis) return;
        const utterance = new SpeechSynthesisUtterance(text);
        const voices = window.speechSynthesis.getVoices();
        let selectedVoice = voices.find(v => /^Google US English/i.test(v.name) && v.lang === 'en-US') ||
                            voices.find(v => v.lang === 'en-US');
        if (selectedVoice) {
            utterance.voice = selectedVoice;
            utterance.lang = selectedVoice.lang;
        } else {
            utterance.lang = 'en-US';
        }
        utterance.rate = 0.95;
        window.speechSynthesis.speak(utterance);
    }

    // ================= Notion 增量同步算法 (SWR) =================

    // 封装通用 Notion API 请求
    function requestNotion(url, method = 'POST', data = null) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method,
                url,
                headers: {
                    'Authorization': `Bearer ${CONFIG.notionToken}`,
                    'Notion-Version': '2022-06-28',
                    'Content-Type': 'application/json'
                },
                data: data ? JSON.stringify(data) : null,
                onload: res => {
                    if (res.status >= 200 && res.status < 300) {
                        try {
                            resolve(JSON.parse(res.responseText));
                        } catch (e) {
                            resolve(res.responseText);
                        }
                    } else {
                        reject(new Error(`HTTP ${res.status}`));
                    }
                },
                onerror: reject
            });
        });
    }

    // SWR 核心逻辑：读取缓存并渲染 -> 后台发起增量请求
    async function syncVocabData() {
        if (isCurrentSiteExcluded()) {
            clearPageHighlights();
            return;
        }
        const cachedDataStr = GM_getValue(CACHE_KEYS.data);
        const lastSyncTime = GM_getValue(CACHE_KEYS.syncTime) || 0;
        const now = Date.now();

        // 1. 如果有缓存且未过期，极速使用缓存
        if (cachedDataStr && (now - lastSyncTime < CONFIG.cacheTTL)) {
            const cachedWords = JSON.parse(cachedDataStr);
            currentMaxNo = GM_getValue(CACHE_KEYS.maxNo) || 0;
            noColumnName = GM_getValue(CACHE_KEYS.colName) || null;

            vocabMap = new Map(cachedWords.map(item => [item.word, item]));
            globalRegex = buildRegex([...vocabMap.keys()]);

            if (globalRegex) {
                scanPage(globalRegex);
                observeDynamicContent(globalRegex);
            }

            console.log(`[LexiPro] ⚡ 极速加载缓存完成: ${vocabMap.size} 词。开始后台增量检查...`);

            // 2. 异步触发后台增量拉取，不阻塞当前页面
            fetchDeltaUpdates(lastSyncTime, cachedWords);

        } else {
            // 没有缓存时，首次全量拉取
            console.log('[LexiPro] ⏳ 首次使用或无缓存，全量拉取 Notion 数据...');
            const allWords = await fetchFromNotion(null);
            updateLocalState(allWords);
        }
    }

    // 增量拉取逻辑 (Delta Fetch)
    async function fetchDeltaUpdates(lastSyncTime, currentCachedWords) {
        // 构建过滤器：只拉取修改时间大于上次同步时间的数据
        const filter = {
            "timestamp": "last_edited_time",
            "last_edited_time": { "after": new Date(lastSyncTime).toISOString() }
        };

        try {
            const updatedWords = await fetchFromNotion(filter);

            if (updatedWords.length > 0) {
                console.log(`[LexiPro] 🔄 后台增量发现 ${updatedWords.length} 个新变动。`);

                // 将新变动的数据合入内存 Map
                updatedWords.forEach(item => vocabMap.set(item.word, item));

                const updatedAllWords = Array.from(vocabMap.values());

                // 保存更新后的全量数据到本地缓存
                GM_setValue(CACHE_KEYS.data, JSON.stringify(updatedAllWords));
                GM_setValue(CACHE_KEYS.syncTime, Date.now());
                GM_setValue(CACHE_KEYS.maxNo, currentMaxNo);
                if (noColumnName) GM_setValue(CACHE_KEYS.colName, noColumnName);

                // 重新挂载高亮正则，点亮刚才在后台拉取的新词
                globalRegex = buildRegex([...vocabMap.keys()]);
                if (globalRegex) scanPage(globalRegex);

            } else {
                // 如果没有新变动，仅更新同步时间戳
                GM_setValue(CACHE_KEYS.syncTime, Date.now());
            }
        } catch(e) {
            console.error('[LexiPro] ⚠️ 后台增量同步失败', e);
        }
    }

    // 拉取执行逻辑 (支持分页和限流控制)
    async function fetchFromNotion(filterObj) {
        const fetchedWords = [];
        let hasMore = true;
        let nextCursor = undefined;
        const url = `https://api.notion.com/v1/databases/${CONFIG.notionDatabaseId}/query`;

        while (hasMore) {
            const body = {};
            if (nextCursor) body.start_cursor = nextCursor;
            if (filterObj) body.filter = filterObj;

            const res = await requestNotion(url, 'POST', body);

            if (res && res.results) {
                for (let page of res.results) {
                    const wordArr = page.properties['Words']?.title;
                    const meaningArr = page.properties['Meaning']?.rich_text;

                    const word = wordArr && wordArr.length > 0 ? wordArr[0].plain_text : '';
                    if (!word) continue;
                    const meaning = meaningArr && meaningArr.length > 0 ? meaningArr[0].plain_text : '暂无释义';

                    // 动态探测自增列列名
                    if (!noColumnName) {
                        const possibleNames = ['# ID', 'ID', '# No', 'No', 'id'];
                        noColumnName = possibleNames.find(name => page.properties[name]) || null;
                    }

                    // 更新最大的自增序号
                    if (noColumnName && page.properties[noColumnName]) {
                        const noVal = page.properties[noColumnName]?.number;
                        if (typeof noVal === 'number' && noVal > currentMaxNo) {
                            currentMaxNo = noVal;
                        }
                    }

                    fetchedWords.push({
                        word: normalizeWord(word),
                        originalWord: word,
                        exp: meaning,
                        id: page.id
                    });
                }
            }

            hasMore = res.has_more;
            nextCursor = res.next_cursor;

            // Notion 速率限制：每秒3次，设定350ms的安全延迟间隔
            if (hasMore) await new Promise(r => setTimeout(r, 350));
        }

        return fetchedWords;
    }

    // 更新本地内存与缓存的状态
    function updateLocalState(allWords) {
        vocabMap = new Map(allWords.map(item => [item.word, item]));
        GM_setValue(CACHE_KEYS.data, JSON.stringify(allWords));
        GM_setValue(CACHE_KEYS.syncTime, Date.now());
        GM_setValue(CACHE_KEYS.maxNo, currentMaxNo);
        if (noColumnName) GM_setValue(CACHE_KEYS.colName, noColumnName);

        globalRegex = buildRegex([...vocabMap.keys()]);
        if (globalRegex) {
            scanPage(globalRegex);
            observeDynamicContent(globalRegex);
        }
    }

    function saveVocabToCache() {
        const allWords = Array.from(vocabMap.values());
        GM_setValue(CACHE_KEYS.data, JSON.stringify(allWords));
    }

    // ================= 页面文本高亮系统 =================

    // 构建用于匹配文本节点的正则表达式
    function buildRegex(words) {
        const escaped = words
            .filter(Boolean)
            .sort((a, b) => b.length - a.length)
            .map(w => w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));

        if (!escaped.length) return null;
        return new RegExp(`\\b(${escaped.join('|')})\\b`, 'gi');
    }

    // 判断该 DOM 节点是否需要跳过（比如本身就在高亮块里，或者是 script 标签）
    function shouldSkipNode(node) {
        const parent = node.parentNode;
        if (!parent || SKIP_TAGS.has(parent.nodeName)) return true;
        if (parent.closest && parent.closest(`.${CONFIG.highlightClass}`)) return true;
        if (!node.nodeValue || !/[A-Za-z]/.test(node.nodeValue)) return true;
        return false;
    }

    function isFlexOrGridParent(element) {
        if (!element || element.nodeType !== Node.ELEMENT_NODE) return false;
        try {
            const display = window.getComputedStyle(element).display || '';
            return display.includes('flex') || display.includes('grid');
        } catch (e) {
            return false;
        }
    }

    // 对特定的文本节点执行正则匹配与高亮 span 的替换
    function highlightTextNode(textNode, regex) {
        if (!regex || shouldSkipNode(textNode)) return;

        const text = textNode.nodeValue;
        regex.lastIndex = 0;

        if (!regex.test(text)) return;
        regex.lastIndex = 0;

        const frag = document.createDocumentFragment();
        let lastIndex = 0;
        let match;

        while ((match = regex.exec(text)) !== null) {
            const raw = match[0];
            const key = normalizeWord(raw);

            if (!vocabMap.has(key)) continue;

            if (match.index > lastIndex) {
                frag.appendChild(document.createTextNode(text.slice(lastIndex, match.index)));
            }

            const span = document.createElement('span');
            span.className = CONFIG.highlightClass;
            span.textContent = raw;
            span.dataset.word = key;

            // 挂载移入与移出的交互事件
            span.addEventListener('mouseenter', handleHighlightEnter);
            span.addEventListener('mouseleave', handleHighlightLeave);

            frag.appendChild(span);
            lastIndex = regex.lastIndex;
        }

        if (lastIndex < text.length) {
            frag.appendChild(document.createTextNode(text.slice(lastIndex)));
        }

        const parent = textNode.parentNode;
        if (!parent) return;

        if (isFlexOrGridParent(parent)) {
            const wrapper = document.createElement('span');
            wrapper.className = 'notion-vocab-inline-container';
            wrapper.appendChild(frag);
            parent.replaceChild(wrapper, textNode);
        } else {
            parent.replaceChild(frag, textNode);
        }
    }

    // 全局扫描页面
    function scanPage(regex, root = document.body) {
        if (isCurrentSiteExcluded()) return;
        if (!regex) return;

        const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, {
            acceptNode(node) {
                return shouldSkipNode(node) ? NodeFilter.FILTER_REJECT : NodeFilter.FILTER_ACCEPT;
            }
        });

        const nodes = [];
        while (walker.nextNode()) nodes.push(walker.currentNode);

        for (const node of nodes) {
            highlightTextNode(node, regex);
        }
    }

    // 观察器：监控动态插入的页面内容（例如无限滚动瀑布流）
    function observeDynamicContent(regex) {
        if (isCurrentSiteExcluded()) return;
        if (vocabObserver) {
            try { vocabObserver.disconnect(); } catch (e) {}
            vocabObserver = null;
        }
        vocabObserver = new MutationObserver(mutations => {
            for (const mutation of mutations) {
                for (const node of mutation.addedNodes) {
                    if (node.nodeType === Node.TEXT_NODE) {
                        highlightTextNode(node, regex);
                    } else if (node.nodeType === Node.ELEMENT_NODE && !SKIP_TAGS.has(node.nodeName)) {
                        scanPage(regex, node);
                    }
                }
            }
        });

        vocabObserver.observe(document.body, { childList: true, subtree: true });
    }

    function showHighlightPopup(target, word) {
        clearTimeout(highlightPopupHideTimeout);

        const key = normalizeWord(word);
        quickTooltip.dataset.word = key;
        quickTooltip.dataset.lookupText = key;
        quickContentBox.innerHTML = renderQuickDict(word, { syncMainState: false });
        updateHighlightPopupStarUI(key);
        quickTooltip.style.display = 'block';

        const rect = target.getBoundingClientRect();
        let x = rect.left + window.scrollX;
        let y = rect.bottom + window.scrollY + 2;

        const tRect = quickTooltip.getBoundingClientRect();
        if (x + tRect.width > window.innerWidth + window.scrollX - 8) {
            x = window.innerWidth + window.scrollX - tRect.width - 8;
        }
        if (y + tRect.height > window.innerHeight + window.scrollY - 8) {
            // 下方空间不足时翻到词上方，箭头同步翻转朝下
            y = rect.top + window.scrollY - tRect.height - 2;
            quickTooltip.classList.add('imt-flip-up');
        } else {
            quickTooltip.classList.remove('imt-flip-up');
        }

        quickTooltip.style.left = `${Math.max(8, x)}px`;
        quickTooltip.style.top = `${Math.max(8, y)}px`;

        if (CONFIG.autoPronounce) {
            playHighlightedWordAudio(word);
        }
        requestAnimationFrame(() => quickTooltip.style.opacity = '1');
    }

            hideTimeout = setTimeout(() => closeTooltip(), 80);
        }
    }

    // ================= Notion 收藏 / 取消收藏逻辑 =================
    function getMeaningToSave(word, sourceText) {
        const cached = translationCache.get(sourceText) || translationCache.get(word);

        // 智能抓取并提取大模型生成的释义
        if (cached && cached.entries && cached.entries.length > 0) {
            const entry = cached.entries[0];
            if (cached.mode === 'en_word') {
                return entry.pos ? `${entry.pos} ${entry.meaning}` : entry.meaning;
            }
            if (cached.mode === 'zh_word') {
                return entry.pos ? `${entry.pos} ${entry.translation}` : entry.translation;
            }
        }

        const existing = vocabMap.get(normalizeWord(word));
        return existing?.exp || "从网页端收藏（暂无释义）";
    }

    async function setVocabStarState(word, willStar, sourceText = word) {
        const key = normalizeWord(word);

        if (willStar) {
            const meaningToSave = getMeaningToSave(key, sourceText);

            // 准备要推送到 Notion API 的 Properties 结构
            const properties = {
                "Words": { title: [{ text: { content: key } }] },
                "Meaning": { rich_text: [{ text: { content: meaningToSave } }] }
            };

            // 执行 Notion ID 自动递增
            if (noColumnName) {
                currentMaxNo += 1;
                properties[noColumnName] = { number: currentMaxNo };
                GM_setValue(CACHE_KEYS.maxNo, currentMaxNo);
            }

            const data = {
                parent: { database_id: CONFIG.notionDatabaseId },
                properties: properties
            };

            // 推送至数据库
            const res = await requestNotion('https://api.notion.com/v1/pages', 'POST', data);

            // 更新内存中的词汇表，并保存返回的 Page ID 供后续可能的删除操作使用
            vocabMap.set(key, {
                word: key,
                originalWord: sourceText || key,
                exp: meaningToSave,
                id: res.id
            });

            globalRegex = buildRegex([...vocabMap.keys()]);
            scanPage(globalRegex);
        } else {
            // 取消收藏：在 Notion 中将此页移入回收站 (archive)
            const pageId = vocabMap.get(key)?.id;
            if (pageId) {
                await requestNotion(`https://api.notion.com/v1/pages/${pageId}`, 'PATCH', { archived: true });
            }
            vocabMap.delete(key);
            globalRegex = buildRegex([...vocabMap.keys()]);
        }

        // 操作完成后，将内存中的字典立刻同步到本地磁盘缓存
        saveVocabToCache();
    }

    function isGlobalImmersionTranslationPage() {
        try {
            // 严格门控：仅在当前页面已启用沉浸式翻译时允许单字点击查词，通用网页（如 TradingView）默认绝不拦截单字点击
            if (typeof isCurrentSiteImmersionEnabled === 'function' && !isCurrentSiteImmersionEnabled()) {
                return false;
            }

            const hostname = location.hostname.toLowerCase();
            const isImmersionSite = hostname === 'reddit.com' || hostname.endsWith('.reddit.com') ||
                                    hostname === 'reuters.com' || hostname.endsWith('.reuters.com');
            if (isImmersionSite) return true;

            if (CONFIG.isSessionImmersionEnabled) return true;

            const list = Array.isArray(CONFIG.immersionSites) ? CONFIG.immersionSites : [];
            if (list.some(pat => matchesDomainPattern(hostname, pat))) return true;

            if (
                Boolean(window.__lexipro_site_showing_translation) ||
                Boolean(window.__lexipro_site_translating) ||
                Boolean(document.querySelector('[data-lexipro-site-id]')) ||
                Boolean(document.querySelector('#lexipro-site-fab-root'))
            ) {
                return true;
            }
        } catch (e) {
            return false;
        }
        return false;
    }

    function extractWordFromTextOffset(text, offset) {
        if (!text || offset < 0) return null;
        let charIndex = offset;
        if (charIndex >= text.length) charIndex = text.length - 1;
        if (charIndex < 0) charIndex = 0;

        // 字符容差：若当前位置非英文字母，尝试左右移动 1 字符命中
        if (!/[a-zA-Z]/.test(text[charIndex])) {
            if (charIndex > 0 && /[a-zA-Z]/.test(text[charIndex - 1])) {
                charIndex--;
            } else if (charIndex + 1 < text.length && /[a-zA-Z]/.test(text[charIndex + 1])) {
                charIndex++;
            } else {
                return null;
            }
        }

        let start = charIndex;
        let end = charIndex;
        while (start > 0 && /[a-zA-Z0-9'-]/.test(text[start - 1])) {
            start--;
        }
        while (end < text.length && /[a-zA-Z0-9'-]/.test(text[end])) {
            end++;
        }

        let rawWord = text.slice(start, end).replace(/^[^a-zA-Z]+|[^a-zA-Z]+$/g, '');
        if (!rawWord || rawWord.length < 2 || !/^[a-zA-Z]+(?:[-'][a-zA-Z]+)*$/.test(rawWord)) {
            return null;
        }

        const realStart = text.indexOf(rawWord, start);
        if (realStart === -1) return null;
        const realEnd = realStart + rawWord.length;

        return {
            word: rawWord,
            start: realStart,
            end: realEnd
        };
    }

    function buildUserPrompt(text, format = null) {
        let prompt = `【Content to Translate】:\n"${text}"`;

        if (format && format.blocks && format.blocks.length > 1) {
            const blockLines = format.blocks
                .map((block, index) => `${index + 1}. [${block.type}] ${block.text}`)
                .join('\n');

            prompt += `\n\nFORMAT_BLOCKS:\n${blockLines}\n\nTranslate each FORMAT_BLOCKS item independently. Return "segments" with exactly ${format.blocks.length} translated strings in the same order.`;
        }

        return prompt;
    }

    function getPreservedSegments(data, originalText, format) {
        if (!format || !format.blocks || format.blocks.length <= 1) return null;

        const expectedCount = format.blocks.length;
        const segments = Array.isArray(data.segments)
            ? data.segments.map(item => String(item || '').trim()).filter(Boolean)
            : [];

        if (segments.length === expectedCount) return segments;

        const translationLines = String(data.translation || '')
            .split(/\r?\n+/)
            .map(line => line.trim())
            .filter(Boolean);

        return translationLines.length === expectedCount ? translationLines : null;
    }

    function renderPreservedBlocks(blocks, segments) {
        const allListItems = blocks.every(block => block.type === 'li');

        if (allListItems) {
            return `<ul class="imt-translation imt-preserved-list">${segments.map(segment => `<li>${escapeHtml(segment)}</li>`).join('')}</ul>`;
        }

        const html = segments.map((segment, index) => {
            const type = blocks[index]?.type || 'line';
            const safeSegment = escapeHtml(segment);

            if (type === 'li') return `<div class="imt-preserved-line">• ${safeSegment}</div>`;
            if (/^h[1-6]$/.test(type)) return `<div class="imt-preserved-heading">${safeSegment}</div>`;
            if (type === 'p' || type === 'blockquote') return `<div class="imt-preserved-paragraph">${safeSegment}</div>`;
            return `<div class="imt-preserved-line">${safeSegment}</div>`;
        }).join('');

        return `<div class="imt-translation imt-preserved-blocks">${html}</div>`;
    }

    function renderSentenceTranslation(data, originalText, format = null) {
        const segments = getPreservedSegments(data, originalText, format);
        if (segments) return renderPreservedBlocks(format.blocks, segments);

        return `<div class="imt-translation">${escapeHtml(data.translation || String(data))}</div>`;
    }

    // 在容器内逐字“打”出内容，保留其内部所有 HTML 结构。speed: 每字间隔 ms。
    function typewriterReveal(container, { speed = 22 } = {}) {
        typewriterToken++;
        const myToken = typewriterToken;

        // 收集全部文本节点（跳过发音喇叭，让其立即出现）
        const walker = document.createTreeWalker(container, NodeFilter.SHOW_TEXT, {
            acceptNode(node) {
                if (!node.nodeValue) return NodeFilter.FILTER_REJECT;
                if (node.parentElement && node.parentElement.closest('.imt-speaker')) {
                    return NodeFilter.FILTER_REJECT;
                }
                return NodeFilter.FILTER_ACCEPT;
            }
        });

        const nodes = [];
        while (walker.nextNode()) nodes.push(walker.currentNode);
        if (!nodes.length) return;

        // 先记录完整文本再清空，准备逐字回填
        const fullText = nodes.map(node => node.nodeValue);
        nodes.forEach(node => { node.nodeValue = ''; });

        let nodeIndex = 0;
        let charPos = 0;

        function step() {
            if (myToken !== typewriterToken) return; // 已被新的请求中断

            // 跳过已显示完的节点，定位到下一个待显示字符
            while (nodeIndex < nodes.length && charPos >= fullText[nodeIndex].length) {
                nodeIndex++;
                charPos = 0;
            }
            if (nodeIndex >= nodes.length) return;

            charPos++;
            nodes[nodeIndex].nodeValue = fullText[nodeIndex].slice(0, charPos);
            container.scrollTop = container.scrollHeight; // 跟随打字自动滚动
            setTimeout(step, speed);
        }

        step();
    }

                || payload?.error?.msg
                || payload?.message
                || payload?.msg

            return detail ? String(detail).slice(0, 300) : '';

        }

    function refreshKimiAccessToken() {
        return new Promise((resolve, reject) => {
            const refreshToken = GM_getValue('lexipro_kimi_refresh_token', '') || CONFIG.llmProviders.kimi_free?.refreshToken;
            if (!refreshToken) {
                reject(new Error('未配置 Kimi Refresh Token，无法自动刷新凭证'));
                return;
            }

            const req = GM_xmlhttpRequest({
                method: 'GET',
                url: 'https://kimi.moonshot.cn/api/auth/token/refresh',
                headers: getKimiHeaders(refreshToken),
                timeout: 20000,
                onload: function(response) {
                    if (response.status === 200) {
                        try {
                            const data = JSON.parse(response.responseText);
                            if (data.access_token) {
                                CONFIG.llmProviders.kimi_free.accessToken = data.access_token;
                                GM_setValue('lexipro_kimi_access_token', data.access_token);
                                if (data.refresh_token) {
                                    CONFIG.llmProviders.kimi_free.refreshToken = data.refresh_token;
                                    GM_setValue('lexipro_kimi_refresh_token', data.refresh_token);
                                }
                                resolve(data.access_token);
                                return;
                            }
                        } catch (e) {}
                    }
                    let errorMsg = `Kimi Token 刷新失败 (HTTP ${response.status})`;
                    try {
                        const errJson = JSON.parse(response.responseText);
                        if (errJson.message || errJson.msg) errorMsg += `：${errJson.message || errJson.msg}`;
                    } catch(e) {}
                    reject(new Error(`${errorMsg}，请在设置中重新填写最新 Token`));
                },
                onerror: function() {
                    reject(new Error('Kimi Token 刷新网络连接出错'));
                },
                ontimeout: function() {
                    reject(new Error('Kimi Token 刷新超时'));
                },
                onabort: function() {
                    reject(new Error('Kimi 请求已取消'));
                }
            });
            activeFetchRequest = req;
        });
    }

                    const suffix = errorDetail ? `：${errorDetail}` : '';

    async function requestKimiWebCompletion({ messages, timeout = 60000 }) {
        getActiveLLMProvider();

        let accessToken = GM_getValue('lexipro_kimi_access_token', '') || CONFIG.llmProviders.kimi_free?.accessToken;
        const refreshToken = GM_getValue('lexipro_kimi_refresh_token', '') || CONFIG.llmProviders.kimi_free?.refreshToken;

        if (!accessToken && refreshToken) {
            accessToken = await refreshKimiAccessToken();
        }
        if (!accessToken) {
            throw new Error('未检测到有效 Kimi Access Token，请在设置中填写');
        }

        let chatResult = await createKimiChat(accessToken);
        if (!chatResult.ok && chatResult.status === 401 && refreshToken) {
            accessToken = await refreshKimiAccessToken();
            chatResult = await createKimiChat(accessToken);
        }

        if (!chatResult.ok) {
            throw new Error(chatResult.error || '创建 Kimi 会话失败');
        }

        const chatId = chatResult.chatId;
        const combinedPrompt = messages.map(m => m.content).filter(Boolean).join('\n\n');
        const kimiMessages = [{ role: 'user', content: combinedPrompt }];

        return new Promise((resolve, reject) => {
            let processedLength = 0;
            let sseBuffer = '';
            let accumulatedText = '';
            let isStreamDone = false;

            function parseSSE(chunk) {
                sseBuffer += chunk;
                const lines = sseBuffer.split('\n');
                sseBuffer = lines.pop();
                for (const rawLine of lines) {
                    const line = rawLine.trim();
                    if (!line.startsWith('data:')) continue;
                    const dataStr = line.slice(5).trim();
                    if (!dataStr || dataStr === '[DONE]') {
                        if (dataStr === '[DONE]') isStreamDone = true;
                        continue;
                    }
                    try {
                        const parsed = JSON.parse(dataStr);
                        if (parsed.event === 'cmpl' && typeof parsed.text === 'string') {
                            accumulatedText += parsed.text;
                        } else if (parsed.event === 'all_done') {
                            isStreamDone = true;
                        } else if (parsed.error || parsed.event === 'error') {
                            const errText = parsed.error?.message || parsed.message || 'Kimi 流式生成返回错误';
                            throw new Error(errText);
                        }
                    } catch (e) {
                        if (e.message && !e.message.includes('Unexpected token') && !e.message.includes('JSON')) {
                            throw e;
                        }
                    }
                }
            }

            const req = GM_xmlhttpRequest({
                method: 'POST',
                url: `https://kimi.moonshot.cn/api/chat/${chatId}/completion/stream`,
                headers: getKimiHeaders(accessToken),
                data: JSON.stringify({
                    messages: kimiMessages,
                    refs: [],
                    use_search: !Boolean(CONFIG.disableThinking),
                    user_search: !Boolean(CONFIG.disableThinking)
                }),
                timeout,
                onprogress: function(response) {
                    const currentText = response.responseText || '';
                    const newChunk = currentText.slice(processedLength);
                    processedLength = currentText.length;
                    try {
                        parseSSE(newChunk);
                    } catch (err) {
                        reject(err);
                    }
                },
                onload: function(response) {
                    if (response.status >= 200 && response.status < 300) {
                        const currentText = response.responseText || '';
                        const newChunk = currentText.slice(processedLength);
                        try {
                            parseSSE(newChunk);
                            parseSSE('\n');
                            if (!accumulatedText.trim()) {
                                throw new Error('Kimi 未返回有效翻译文本');
                            }
                            resolve(accumulatedText);
                        } catch (err) {
                            reject(err);
                        }
                    } else {
                        let errDetail = '';
                        try {
                            const errJson = JSON.parse(response.responseText);
                            errDetail = errJson.message || errJson.msg || '';
                        } catch(e) {}
                        const suffix = errDetail ? `：${errDetail}` : '';
                        reject(new Error(`Kimi 翻译流请求失败 (HTTP ${response.status})${suffix}`));
                    }
                },
                onerror: function() {
                    reject(new Error('Kimi 翻译请求网络连接异常'));
                },
                ontimeout: function() {
                    reject(new Error(`Kimi 在 ${Math.round(timeout / 1000)} 秒内未完成响应，请稍后重试`));
                },
                onabort: function() {
                    reject(new Error('Kimi 请求已取消'));
                }
            });
            activeFetchRequest = req;
        });
    }

    function extractJsonFromContent(content) {
        if (typeof content !== 'string') return content;
        let cleanStr = content.replace(/```json\s*/gi, '').replace(/```\s*/g, '').trim();
        try {
            return JSON.parse(cleanStr);
        } catch (e) {
            const firstBrace = cleanStr.indexOf('{');
            const lastBrace = cleanStr.lastIndexOf('}');
            if (firstBrace !== -1 && lastBrace > firstBrace) {
                return JSON.parse(cleanStr.slice(firstBrace, lastBrace + 1));
            }
            throw e;
        }
    }

    // 页面级翻译请求独立管理器（与划词查词请求完全隔离，避免划词互踩取消）
    const activePageRequests = new Set();

    function requestLLMChatCompletion({
        messages,
        temperature = 0.1,
        responseFormat = null,
        timeout = 45000,
        maxTokens = 2048,
        modelOverride = null,
        requestScope = 'selection'
    }) {
        const providerKey = CONFIG.llmProvider || 'siliconflow';
        if (providerKey === 'kimi_free') {
            return requestKimiWebCompletion({ messages, timeout });
        }

        return new Promise((resolve, reject) => {
            let provider;
            try {
                provider = getActiveLLMProvider();
            } catch (e) {
                reject(e);
                return;
            }

            const body = {
                model: modelOverride || provider.model,
                messages,
                temperature,
                stream: false,
                max_tokens: maxTokens
            };

            if (responseFormat && provider.supportsJsonResponse !== false) {
                body.response_format = responseFormat;
            }

            const shouldDisableThinking = typeof CONFIG.disableThinking === 'boolean'
                ? CONFIG.disableThinking
                : Boolean(provider.disableThinking);

            if (shouldDisableThinking) {
                body.enable_thinking = false;
                body.thinking = { type: 'disabled' };
            } else {
                body.enable_thinking = true;
                body.thinking = { type: 'enabled' };
            }

            let reqHandle = null;
            const cleanupReq = () => {
                if (requestScope === 'page' && reqHandle) {
                    activePageRequests.delete(reqHandle);
                } else if (activeFetchRequest === reqHandle) {
                    activeFetchRequest = null;
                }
            };

            reqHandle = GM_xmlhttpRequest({
                method: "POST",
                url: provider.apiUrl,
                timeout,
                headers: {
                    "Content-Type": "application/json",
                    "Authorization": `Bearer ${provider.apiKey}`
                },
                data: JSON.stringify(body),
                onload: function(response) {
                    cleanupReq();
                    if (response.status >= 200 && response.status < 300) {
                        try {
                            const resJson = JSON.parse(response.responseText);
                            const choice = resJson.choices?.[0];
                            if (choice?.finish_reason === 'length') {
                                throw new Error(`输出达到 ${maxTokens} tokens 上限`);
                            }
                            const content = choice?.message?.content;
                            if (!content) {
                                throw new Error("响应中缺少 choices[0].message.content");
                            }
                            resolve(content);
                        } catch (e) {
                            reject(new Error(`解析 ${provider.name} 响应失败: ${e.message}`));
                        }
                    } else {
                        const detail = getLLMErrorDetail(response);
                        const suffix = detail ? `：${detail}` : '';
                        reject(new Error(`${provider.name} API 请求失败（HTTP ${response.status}）${suffix}`));
                    }
                },
                onerror: function() {
                    cleanupReq();
                    reject(new Error(`${provider.name} 网络连接出错`));
                },
                ontimeout: function() {
                    cleanupReq();
                    reject(new Error(`${provider.name} 在 ${Math.round(timeout / 1000)} 秒内未完成响应，请稍后重试`));
                },
                onabort: function() {
                    cleanupReq();
                    reject(new Error(`${provider.name} 请求已取消`));
                }
            });

            if (requestScope === 'page') {
                activePageRequests.add(reqHandle);
            } else {
                activeFetchRequest = reqHandle;
            }
        });
    }

    // 发起真实的 API Post 请求
    function fetchTranslation(text, targetBox = contentBox, { syncMainState = true, format = null } = {}) {
        // 如果有上一次正在进行的请求，立即中断取消
        abortActiveFetch();

        currentRequestId++;
        const requestId = currentRequestId;
        isRequesting = true;

        requestLLMChatCompletion({
            timeout: 45000,
            maxTokens: 2048,
            responseFormat: { type: "json_object" },
            temperature: 0.1,
            messages: [
                { role: "system", content: SYSTEM_PROMPT },
                { role: "user", content: buildUserPrompt(text, format) }
            ]
        }).then(content => {
            if (requestId !== currentRequestId) return;
            isRequesting = false;
            activeFetchRequest = null;
            try {
                const resultData = extractJsonFromContent(content);

                // 存入内存翻译缓存
                translationCache.set(text, resultData);
                targetBox.innerHTML = renderTranslation(resultData, text, { syncMainState, format });
                typewriterReveal(targetBox);
                if (!syncMainState) {
                    quickTooltip.dataset.word = normalizeWord(resultData.word || text);
                    updateHighlightPopupStarUI(quickTooltip.dataset.word);
                }

                // 单词模式下仅在开启自动发音设置时自动朗读
                if (CONFIG.autoPronounce && resultData.mode === "en_word") playAudio(resultData.word || text);
            } catch (e) {
                targetBox.innerHTML = `<div style="color:#ef746f;font-weight:normal;">解析失败: ${escapeHtml(e.message)}</div>`;
            }
        }).catch(e => {
            if (requestId !== currentRequestId) return;
            isRequesting = false;
            activeFetchRequest = null;
            targetBox.innerHTML = `<div style="color:#ef746f;font-weight:normal;">${escapeHtml(e.message)}</div>`;
        });
    }

    function isTextEditableElement(target) {
        if (!target) return false;
        if (target.isContentEditable) return true;

        const tagName = target.tagName;
        if (tagName === 'TEXTAREA') return true;
        if (tagName === 'INPUT') {
            const type = (target.type || 'text').toLowerCase();

    function translateText(text) {
        // 精确判定输入语言：若包含中文字符，则定向翻译为英文；反之定向翻译为中文
        const hasChinese = /[\u4e00-\u9fa5\u3000-\u303f\uff01-\uff0f\uff1a-\uff20\uff3b-\uff40\uff5b-\uff65]/.test(text);

        let systemPrompt;
        if (hasChinese) {
            // 中文 -> 英文（强约束：只做翻译，即使是问句也绝不回答问题）
            systemPrompt = `You are a professional translator. The user entered Chinese text. Your SOLE task is to translate it into natural, fluent, and idiomatic English.
CRITICAL RULES:
1. Translate to English ONLY.
2. Even if the input is a question (e.g. "为什么不..."), do NOT answer the question! You must translate the question sentence itself into English.
3. Return ONLY the translated English sentence. No quotes, no markdown fences, no explanatory prefix or suffix.`;
        } else {
            // 英文 -> 中文（强约束：只做翻译，严禁回答问题）
            systemPrompt = `你是一名专业双语翻译官。用户输入的是英文内容，你的唯一任务是将其翻译为地道、自然的简体中文。
严格要求：
1. 必须且仅翻译为中文。即使输入是一个问题，也绝对不要去回答问题本身，必须仅翻译该句子。
2. 仅输出最终的中文译文，不要引号，不要 markdown 格式，不要多余的前缀、后缀或解释。`;
        }

        return requestLLMChatCompletion({
            timeout: 30000,
            maxTokens: 2048,
            temperature: 0.2,
            messages: [
                { role: "system", content: systemPrompt },
                { role: "user", content: text }
            ]
        }).then(content => {
            content = content.trim();
            // 移除可能存在的 markdown 代码块包裹
            if (content.startsWith('```') && content.endsWith('```')) {
                content = content.replace(/^```[a-zA-Z]*\n?/, '').replace(/\n?```$/, '').trim();
            }
            // 移除首尾引号
            if ((content.startsWith('"') && content.endsWith('"')) || (content.startsWith("'") && content.endsWith("'"))) {
                content = content.slice(1, -1).trim();
            }
            return content.trim();
        }).catch(e => {
            throw new Error("解析翻译响应失败: " + e.message);
        });
    }

    function replaceElementContent(target, newText) {
        target.focus();
        let success = false;
        try {
            if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA') {
                target.select();
                success = document.execCommand('insertText', false, newText);
                if (!success) {
                    setNativeInputValue(target, newText);
                }
            } else if (target.isContentEditable) {
                const range = document.createRange();
                range.selectNodeContents(target);
                const sel = window.getSelection();
                sel.removeAllRanges();
                sel.addRange(range);
                success = document.execCommand('insertText', false, newText);
                if (!success) {
                    target.innerHTML = '';
                    target.appendChild(document.createTextNode(newText));
                }
            }
        } catch (e) {
            console.error('[LexiPro] execCommand 替换失败，采用原生赋值兜底:', e);
            if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA') {
                setNativeInputValue(target, newText);
            } else if (target.isContentEditable) {
                target.innerText = newText;
            }
        }

        // 统一派发 React/Lit/原生事件通知输入框状态更新
        target.dispatchEvent(new InputEvent('input', { bubbles: true, cancelable: true, inputType: 'insertText', data: newText }));
        target.dispatchEvent(new Event('input', { bubbles: true }));
        target.dispatchEvent(new Event('change', { bubbles: true }));

        // 光标定位至文本末尾
        try {
            if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA') {
                if (typeof target.setSelectionRange === 'function') {
                    target.setSelectionRange(target.value.length, target.value.length);
                }
            } else if (target.isContentEditable) {
                const range = document.createRange();
                range.selectNodeContents(target);
                range.collapse(false);
                const sel = window.getSelection();
                sel.removeAllRanges();
                sel.addRange(range);
            }
        } catch (posErr) {}
    }

    // 兼容 React / Vue 受控组件的原生 setter
    function setNativeInputValue(element, value) {
        const valueSetter = Object.getOwnPropertyDescriptor(element, 'value')?.set;
        const prototype = Object.getPrototypeOf(element);
        const prototypeValueSetter = Object.getOwnPropertyDescriptor(prototype, 'value')?.set;

        if (prototypeValueSetter && valueSetter !== prototypeValueSetter) {
            prototypeValueSetter.call(element, value);
        } else if (valueSetter) {
            valueSetter.call(element, value);
        } else {
            element.value = value;
        }
    }

    async function handleInlineTranslation(target) {
        // 定位真正的可编辑目标节点（如在外层 contenteditable 容器内的子节点）
        const editable = (target.isContentEditable ? (target.closest('[contenteditable="true"]') || target) : target);
        let originalText = '';
        if (editable.tagName === 'INPUT' || editable.tagName === 'TEXTAREA') {
            originalText = editable.value;
        } else if (editable.isContentEditable) {
            originalText = editable.innerText || editable.textContent || '';
        }

        const textToTranslate = originalText.replace(/[\s\u00A0]+$/, '');
        if (!textToTranslate) return;

        editable.dataset.translating = 'true';
        showInputLoading(editable);

        try {
            const translated = await translateText(textToTranslate);
            if (translated) {
                replaceElementContent(editable, translated);
            }
        } catch (err) {
            console.error('[LexiPro] 连续空格键自动翻译出错:', err);
            // 发生异常时清理刚才输入的连续空格
            replaceElementContent(editable, textToTranslate);
        } finally {
            editable.dataset.translating = 'false';
            hideInputLoading(editable);
        }
    }

    document.addEventListener('keydown', function(e) {
        if (e.key === ' ' || e.code === 'Space') {
            const target = e.target;
            if (!isTextEditableElement(target)) return;

            if (target.dataset.translating === 'true') {
                e.preventDefault();
                return;
            }

            const now = Date.now();
            spacePressHistory = spacePressHistory.filter(t => now - t < 800);
            spacePressHistory.push(now);

            if (spacePressHistory.length >= 3) {
                spacePressHistory = [];
                e.preventDefault();
                handleInlineTranslation(target);
            }
        } else {
            spacePressHistory = [];
        }
    }, true);

        // ================= 借鉴开源沉浸式翻译 (Immersive Translate) 核心算法与设计 =================
        // 沉浸式翻译开源规范：块级语义元素标签白名单
        const IMMERSIVE_BLOCK_TAGS = new Set([
            'P', 'H1', 'H2', 'H3', 'H4', 'H5', 'H6',
            'LI', 'BLOCKQUOTE', 'ARTICLE', 'SECTION',
            'DD', 'DT', 'FIGCAPTION'
        ]);

        // 沉浸式翻译开源规范：排除不可翻译或干扰排版的标签黑名单
        const IMMERSIVE_EXCLUDE_TAGS = new Set([
            'SCRIPT', 'STYLE', 'CODE', 'PRE', 'SVG', 'MATH',
            'NOSCRIPT', 'IFRAME', 'CANVAS', 'VIDEO', 'AUDIO',
            'INPUT', 'TEXTAREA', 'SELECT', 'OPTION', 'BUTTON'
        ]);

        // 判定元素是否为有效待翻译块级节点（启发式可见性与语义层级检查）
        function isTranslatableBlock(el) {
            if (!el || el.nodeType !== Node.ELEMENT_NODE) return false;
            const tag = el.tagName;
            if (IMMERSIVE_EXCLUDE_TAGS.has(tag)) return false;
            if (el.isContentEditable) return false;
            if (el.getAttribute('translate') === 'no' || el.classList.contains('notranslate')) return false;
            if (el.closest('.lexipro-site-fab-container') || el.closest('.immersive-translate-target-translation') || el.closest('#imt-shadow-host')) return false;

            // 检查元素可见性（避免翻译隐藏元素）
            if (el.offsetParent === null && el.offsetWidth === 0 && el.offsetHeight === 0) return false;

            if (IMMERSIVE_BLOCK_TAGS.has(tag)) {
                return true;
            }

            // 对普通 div 或 role="article"/"paragraph"，仅当其不包含块级子节点且具有足够文本量时作为段落
            if (tag === 'DIV' || el.getAttribute('role') === 'article' || el.getAttribute('role') === 'paragraph') {
                const hasBlockChild = el.querySelector('p, h1, h2, h3, h4, h5, h6, article, section, blockquote, li');
                if (!hasBlockChild) {
                    const text = (el.innerText || '').trim();
                    if (text.length >= 20 && /[a-zA-Z]{3,}/.test(text)) {
                        return true;
                    }
                }
            }
            return false;
        }

        // 沉浸式翻译开源规范：有效可翻译文本启发式校验（排除纯数字、代码标识、纯链接或中文主体文本）
        function isValidTranslatableText(text) {
            if (!text) return false;
            const trimmed = text.trim();
            if (trimmed.length < 3) return false;
            if (/^https?:\/\/\S+$/i.test(trimmed)) return false;
            if (/^[\d\s.,;:'"\/\\()\[\]{}#@!%&*+=_-]+$/.test(trimmed)) return false;
            if (!/[a-zA-Z]{2,}/.test(trimmed)) return false;
            const chineseChars = trimmed.match(/[\u4e00-\u9fa5]/g) || [];
            if (chineseChars.length > trimmed.length * 0.6) return false;
            return true;
        }

        // DOM 元素收集（针对专有站点深度定制 + 通用网页全自动语义提取）
        function collectSiteElements() {
            const newlyCollected = [];

            if (siteConfig.isReddit) {
                // 1. 帖子大标题
                document.querySelectorAll('shreddit-post [slot="title"], shreddit-post a[slot="title"], h1[slot="title"], [data-testid="post-title"], div[data-testid="post-container"] h1, p.title a.title').forEach(el => {
                    addSiteItem(el, 'title', newlyCollected);
                });

                // 2. 帖子正文（优先精准提取真正放文本的内部段落/容器，防止破坏外层插槽结构）
                document.querySelectorAll('shreddit-post [slot="text-body"] div[id*="-post-rtjson-content"], shreddit-post [slot="text-body"] > div, shreddit-post [slot="text-body"] p, [data-testid="post-body"], div[data-click-id="text"], div.expando div.usertext-body').forEach(el => {
                    addSiteItem(el, 'body', newlyCollected);
                });
                document.querySelectorAll('shreddit-post [slot="text-body"]').forEach(el => {
                    addSiteItem(el, 'body', newlyCollected);
                });

                // 3. 各级评论
                document.querySelectorAll('shreddit-comment [slot="comment"], shreddit-comment div[id*="-comment-rtjson-content"], div[data-testid="comment"], div.Comment, div.comment div.usertext-body').forEach(el => {
                    addSiteItem(el, 'comment', newlyCollected);
                });
                document.querySelectorAll('shreddit-comment p').forEach(el => {
                    if (el.closest('[slot="comment"]') || el.closest('div[id*="-comment-rtjson-content"]')) return;
                    addSiteItem(el, 'comment', newlyCollected);
                });
            } else if (siteConfig.isReuters) {
                document.querySelectorAll('h1[data-testid="Heading"], header h1, h1.article-header__title, article h1').forEach(el => {
                    addSiteItem(el, 'title', newlyCollected);
                });
                document.querySelectorAll('p[data-testid*="paragraph-0"], p[data-testid="Paragraph"], header p').forEach(el => {
                    addSiteItem(el, 'lead', newlyCollected);
                });
                document.querySelectorAll('article div[data-testid*="paragraph-"], article p[data-testid*="paragraph-"], div[class*="article-body__content"] p, div[data-testid="article-body"] p').forEach(el => {
                    const text = (el.innerText || '').trim();
                    if (text.startsWith('Reporting by') || text.startsWith('Editing by') || text.includes('Our Standards:')) return;
                    addSiteItem(el, 'paragraph', newlyCollected);
                });
            }

            // 沉浸式翻译开源设计：通用块级语义段落抓取（适配任意通用网页以及专有站点的补充正文）
            const generalBlocks = document.querySelectorAll(`
                p, h1, h2, h3, h4, h5, h6,
                blockquote, article p, section p, main p,
                div[role="article"] p, div[role="paragraph"],
                li, dt, dd, figcaption
            `);
            generalBlocks.forEach(el => {
                if (!isTranslatableBlock(el)) return;
                const text = (el.innerText || '').trim();
                if (!isValidTranslatableText(text)) return;
                addSiteItem(el, 'general_block', newlyCollected);
            });

            return newlyCollected;
        }

        function addSiteItem(element, type, collectorArray) {
            if (!element || element.dataset.lexiproSiteId) return;
            if (element.closest('.lexipro-site-fab-container') || element.closest('.lexipro-trans-card') || element.closest('.lexipro-trans-underline') || element.closest('#imt-shadow-host')) return;

            // 核心防御：防止父子节点重复收集、互踩覆盖造成帖子正文在滚动后凭空消失
            if (element.closest('[data-lexipro-site-id]')) return;
            if (element.querySelector('[data-lexipro-site-id]')) return;

            const text = (element.innerText || '').trim();
            if (!text || text.length < 2) return;
            if (/^https?:\/\/\S+$/i.test(text)) return;

            const id = 'lpsite_' + (++siteItemCounter) + '_' + Date.now();
            element.dataset.lexiproSiteId = id;

            const item = {
                id,
                type,
                element,
                origWrapper: null,
                transContainer: null,
                originalText: text,
                translatedText: null,
                injectedNode: null,
                isShowingOriginal: false,
                status: 'pending' // pending | translating | done | failed
            };

            siteItemsMap.set(id, item);
            collectorArray.push(item);
        }

        // 视口空间绝对排序：确保页面自顶向下（标题、正文、前排热评）优先出屏
        function sortPendingItemsByVisualOrder(items) {
            return items.slice().sort((a, b) => {
                if (a.type === 'title' && b.type !== 'title') return -1;
                if (b.type === 'title' && a.type !== 'title') return 1;
                if (a.type === 'body' && b.type !== 'body') return -1;
                if (b.type === 'body' && a.type !== 'body') return 1;

                const rectA = a.element ? a.element.getBoundingClientRect() : { top: 0 };
                const rectB = b.element ? b.element.getBoundingClientRect() : { top: 0 };
                const topA = rectA.top + (window.scrollY || 0);
                const topB = rectB.top + (window.scrollY || 0);
                return topA - topB;
            });
        }

        // 批次拆分构造器
        function createBatches(items, maxPerBatch = 6) {
            const batches = [];
            let curBatch = [];
            let curLen = 0;
            for (const it of items) {
                if (it.originalText.length > 350 || it.type === 'body') {
                    if (curBatch.length > 0) {
                        batches.push(curBatch);
                        curBatch = [];
                        curLen = 0;
                    }
                    batches.push([it]);
                } else {
                    curBatch.push(it);
                    curLen += it.originalText.length;
                    if (curBatch.length >= maxPerBatch || curLen > 800) {
                        batches.push(curBatch);
                        curBatch = [];
                        curLen = 0;
                    }
                }
            }
            if (curBatch.length > 0) {
                batches.push(curBatch);
            }
            return batches;
        }

        // 核心渲染逻辑：应用 3 种翻译样式（保留原 DOM 树，杜绝 innerHTML 反序列化破坏事件与状态）
        function applyItemStyle(item) {
            if (!item.element || !item.translatedText) return;

            // 清理已存在的辅助独立节点
            if (item.injectedNode && item.injectedNode.parentNode) {
                item.injectedNode.remove();
                item.injectedNode = null;
            }

            // 初始化或取得原生子节点的包装器（保留所有子节点及绑定在上面的原生事件监听）
            let origWrapper = item.origWrapper;
            if (!origWrapper || !origWrapper.isConnected) {
                origWrapper = item.element.querySelector(':scope > .lexipro-orig-wrapper');
                if (!origWrapper) {
                    origWrapper = document.createElement('span');
                    origWrapper.className = 'lexipro-orig-wrapper';
                    while (item.element.firstChild) {
                        origWrapper.appendChild(item.element.firstChild);
                    }
                    item.element.appendChild(origWrapper);
                }
                item.origWrapper = origWrapper;
            }

            let transContainer = item.transContainer;
            if (!transContainer || !transContainer.isConnected) {
                transContainer = item.element.querySelector(':scope > .lexipro-trans-container');
                item.transContainer = transContainer;
            }

            if (!isSiteShowingTranslation) {
                // 还原状态：显示原生内容，隐藏译文
                item.isShowingOriginal = false;
                if (origWrapper) origWrapper.style.display = '';
                if (transContainer) transContainer.style.display = 'none';
                return;
            }

            if (siteTranslationStyle === 'replace') {
                // 模式 1: 沉浸式中文阅读（原节点保留隐藏，展示中文与切换胶囊）
                if (!transContainer) {
                    transContainer = document.createElement('font');
                    transContainer.className = 'immersive-translate-target-translation lexipro-trans-container';
                    transContainer.setAttribute('data-immersive-translate-translation-element-mark', '_');
                    item.element.appendChild(transContainer);
                    item.transContainer = transContainer;
                }

                transContainer.innerHTML = '';
                const isShowingOrig = item.isShowingOriginal === true;

                origWrapper.style.display = isShowingOrig ? '' : 'none';
                transContainer.style.display = isShowingOrig ? 'inline-flex' : 'inline';

                if (!isShowingOrig) {
                    const textSpan = document.createElement('span');
                    textSpan.className = 'lexipro-immersive-text';
                    textSpan.style.fontFamily = 'inherit';
                    textSpan.style.fontSize = 'inherit';
                    textSpan.style.lineHeight = 'inherit';
                    textSpan.style.color = 'inherit';
                    textSpan.style.fontWeight = 'inherit';

                    if (item.type === 'body' || item.translatedText.includes('\n')) {
                        const paras = item.translatedText.split(/\n\s*\n/).filter(p => p.trim());
                        if (paras.length > 1) {
                            paras.forEach(p => {
                                const pEl = document.createElement('p');
                                pEl.style.margin = '0 0 10px 0';
                                pEl.style.lineHeight = '1.7';
                                pEl.innerText = p.trim();
                                textSpan.appendChild(pEl);
                            });
                        } else {
                            textSpan.innerText = item.translatedText;
                            textSpan.style.whiteSpace = 'pre-line';
                        }
                    } else {
                        textSpan.innerText = item.translatedText;
                    }
                    transContainer.appendChild(textSpan);
                }

                // 微型切换原文/中文胶囊按钮
                const toggleBtn = document.createElement('span');
                toggleBtn.className = 'lexipro-orig-toggle-btn' + (isShowingOrig ? ' showing-orig' : '');
                toggleBtn.setAttribute('role', 'button');
                toggleBtn.setAttribute('tabindex', '0');
                toggleBtn.setAttribute('title', isShowingOrig ? '点击返回中文译文' : '点击查看英文原文');
                toggleBtn.innerHTML = isShowingOrig 
                    ? '<span class="lexipro-orig-toggle-icon">⇄</span> 中文'
                    : '<span class="lexipro-orig-toggle-icon">⇄</span> 原文';

                toggleBtn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    e.preventDefault();
                    item.isShowingOriginal = !item.isShowingOriginal;
                    applyItemStyle(item);
                });

                transContainer.appendChild(toggleBtn);
            } else if (siteTranslationStyle === 'underline') {
                // 模式 2: 下划线双语对照
                if (origWrapper) origWrapper.style.display = '';
                if (transContainer) transContainer.style.display = 'none';

                const under = document.createElement('div');
                under.className = 'immersive-translate-target-translation lexipro-trans-underline';
                under.setAttribute('data-immersive-translate-translation-element-mark', '_');
                under.style.color = 'inherit';
                under.style.fontWeight = 'inherit';
                under.style.fontSize = 'inherit';
                under.style.lineHeight = 'inherit';
                under.style.fontFamily = 'inherit';
                under.style.opacity = '1';
                under.innerText = item.translatedText;
                item.element.appendChild(under);
                item.injectedNode = under;
            } else if (siteTranslationStyle === 'card') {
                // 模式 3: 独立卡片对照
                if (origWrapper) origWrapper.style.display = '';
                if (transContainer) transContainer.style.display = 'none';

                const card = document.createElement('div');
                card.className = 'immersive-translate-target-translation lexipro-trans-card';
                card.setAttribute('data-immersive-translate-translation-element-mark', '_');
                card.innerHTML = `
                    <div class="lexipro-trans-card-header">
                        <span class="lexipro-trans-card-badge">中文译文</span>
                        <button class="lexipro-trans-card-btn" type="button">复制</button>
                    </div>
                    <div class="lexipro-trans-card-body">${escapeHtml(item.translatedText).replace(/\n/g, '<br>')}</div>
                `;
                const copyBtn = card.querySelector('.lexipro-trans-card-btn');
                copyBtn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    navigator.clipboard.writeText(item.translatedText);
                    copyBtn.innerText = '已复制';
                    setTimeout(() => { copyBtn.innerText = '复制'; }, 2000);
                });
                item.element.appendChild(card);
                item.injectedNode = card;
            }
        }

        // 核心翻译调度器（支持失败项单独重试、视口自顶向下流式上屏、请求作用域隔离）
        async function runSiteTranslation({ retryFailedOnly = false } = {}) {
            if (isSiteTranslating) {
                showSiteToast('正在翻译中，请稍候...');
                return;
            }

            collectSiteElements();

            if (retryFailedOnly) {
                // 仅将失败项重新置入待翻译状态，绝不重复翻译已成功的条目
                siteItemsMap.forEach(it => {
                    if (it.status === 'failed' || it.status === 'error') {
                        it.status = 'pending';
                    }
                });
            }

            const pendingItems = Array.from(siteItemsMap.values()).filter(it => it.status === 'pending');

            if (pendingItems.length === 0) {
                const failedCount = Array.from(siteItemsMap.values()).filter(it => it.status === 'failed' || it.status === 'error').length;
                if (siteItemsMap.size === 0) {
                    showSiteToast('未发现可翻译正文');
                } else if (failedCount > 0) {
                    showSiteToast(`当前有 ${failedCount} 条翻译未完成，可点击重试`);
                } else {
                    showSiteToast('本页全部内容均已翻译完成 ✨');
                }
                isSiteShowingTranslation = true;
                siteItemsMap.forEach(it => { if (it.status === 'done') applyItemStyle(it); });
                updateSiteFabState();
                return;
            }

            isSiteTranslating = true;
            isSiteShowingTranslation = true;
            siteTotalCount = siteItemsMap.size;
            siteUntranslatedNewCount = 0;
            updateSiteFabState();

            const thisGen = siteGeneration;

            try {
                // 1. 严格自顶向下视觉排序：保证最上面的帖子标题、正文与前排热评排在最前面
                const sortedItems = sortPendingItemsByVisualOrder(pendingItems);

                // 2. 核心优先队列：提取前 10 条作为第一高优先级批次（用户当前眼前看到的内容）
                const priorityItems = sortedItems.slice(0, 10);
                const remainingItems = sortedItems.slice(10);

                // 前 10 条分拆为小批次并发竞速，完成后立即上屏呈现给用户
                const priorityBatches = createBatches(priorityItems, 5);
                if (priorityBatches.length > 0) {
                    await Promise.all(priorityBatches.map((b, idx) => processSiteBatch(b, idx, thisGen)));
                    if (thisGen !== siteGeneration) return;
                    updateSiteFabState();
                }

                // 3. 后续批次（第 11 条往后）进入后台 Worker 并发池，边翻译边动态上屏
                if (remainingItems.length > 0) {
                    const backgroundBatches = createBatches(remainingItems, 6);
                    let bIdx = 0;
                    const concurrency = Math.min(4, Math.max(2, backgroundBatches.length));
                    const workers = Array.from({ length: concurrency }, async (_, workerId) => {
                        while (bIdx < backgroundBatches.length) {
                            if (thisGen !== siteGeneration) break;
                            const currentBatchIdx = bIdx++;
                            const batch = backgroundBatches[currentBatchIdx];
                            await processSiteBatch(batch, workerId + currentBatchIdx, thisGen);
                            if (thisGen !== siteGeneration) break;
                            updateSiteFabState();
                        }
                    });
                    await Promise.all(workers);
                }

                if (thisGen !== siteGeneration) return;

                const finalFailed = Array.from(siteItemsMap.values()).filter(it => it.status === 'failed' || it.status === 'error').length;
                siteTranslatedCount = Array.from(siteItemsMap.values()).filter(it => it.status === 'done').length;

                if (finalFailed > 0) {
                    showSiteToast(`已翻译 ${siteTranslatedCount} 条，${finalFailed} 条暂未完成（可重试）`);
                } else {
                    showSiteToast(`🎉 成功翻译 ${siteTranslatedCount} 条内容！`);
                }
            } catch (e) {
                console.error('[LexiPro] 网页全文翻译出错:', e);
                showSiteToast('⚠️ 翻译中断: ' + (e.message || e));
            } finally {
                if (thisGen === siteGeneration) {
                    isSiteTranslating = false;
                    updateSiteFabState();
                }
            }
        }

        async function processSiteBatch(batch, seedIndex = 0, gen = siteGeneration) {
            if (!batch || batch.length === 0) return;

            const primaryModel = getModelForBatch(seedIndex);
            const fallbackModel = getModelForBatch(seedIndex + 1);

            if (batch.length === 1) {
                const it = batch[0];
                try {
                    it.status = 'translating';
                    const prompt = `Translate the following ${siteConfig.name} text into natural Simplified Chinese. Return ONLY the translation without quotes, notes, or markdown blocks.\n\n"${it.originalText}"`;
                    let res;
                    try {
                        res = await requestLLMChatCompletion({
                            messages: [
                                { role: "system", content: "You are a professional translator. Return ONLY the translated Chinese text." },
                                { role: "user", content: prompt }
                            ],
                            temperature: 0.2,
                            timeout: 20000,
                            modelOverride: primaryModel,
                            requestScope: 'page'
                        });
                    } catch (err1) {
                        res = await requestLLMChatCompletion({
                            messages: [
                                { role: "system", content: "You are a professional translator. Return ONLY the translated Chinese text." },
                                { role: "user", content: prompt }
                            ],
                            temperature: 0.2,
                            timeout: 20000,
                            modelOverride: fallbackModel,
                            requestScope: 'page'
                        });
                    }
                    if (gen !== siteGeneration) return;
                    it.translatedText = cleanSiteTrans(res);
                    it.status = 'done';
                    siteTranslatedCount = Array.from(siteItemsMap.values()).filter(x => x.status === 'done').length;
                    applyItemStyle(it);
                } catch (e) {
                    if (gen !== siteGeneration) return;
                    it.status = 'failed';
                }
                return;
            }

            try {
                const itemsPayload = batch.map(it => ({ id: it.id, text: it.originalText }));
                const sysPrompt = `Translate these ${siteConfig.name} contents into fluent Simplified Chinese. Return RAW JSON ONLY matching this format:\n{\n  "translations": [\n    { "id": "item_id", "text": "Translated Chinese text" }\n  ]\n}`;
                const userPrompt = JSON.stringify(itemsPayload, null, 2);

                let res;
                try {
                    res = await requestLLMChatCompletion({
                        messages: [
                            { role: "system", content: sysPrompt },
                            { role: "user", content: userPrompt }
                        ],
                        temperature: 0.2,
                        timeout: 30000,
                        modelOverride: primaryModel,
                        requestScope: 'page'
                    });
                } catch (err1) {
                    res = await requestLLMChatCompletion({
                        messages: [
                            { role: "system", content: sysPrompt },
                            { role: "user", content: userPrompt }
                        ],
                        temperature: 0.2,
                        timeout: 30000,
                        modelOverride: fallbackModel,
                        requestScope: 'page'
                    });
                }

                if (gen !== siteGeneration) return;

                let parsed = null;
                try {
                    let cleaned = res.trim();
                    if (cleaned.includes('```json')) cleaned = cleaned.split('```json')[1].split('```')[0].trim();
                    else if (cleaned.includes('```')) cleaned = cleaned.split('```')[1].split('```')[0].trim();
                    parsed = JSON.parse(cleaned);
                } catch (pe) {}

                const map = new Map();
                if (Array.isArray(parsed?.translations)) {
                    parsed.translations.forEach(t => { if (t && t.id && t.text) map.set(t.id, cleanSiteTrans(t.text)); });
                }

                for (const it of batch) {
                    if (gen !== siteGeneration) return;
                    const trans = map.get(it.id);
                    if (trans) {
                        it.translatedText = trans;
                        it.status = 'done';
                        siteTranslatedCount = Array.from(siteItemsMap.values()).filter(x => x.status === 'done').length;
                        applyItemStyle(it);
                    } else {
                        try {
                            const single = await requestLLMChatCompletion({
                                messages: [
                                    { role: "system", content: "Translate to Simplified Chinese. Return ONLY translated text." },
                                    { role: "user", content: it.originalText }
                                ],
                                temperature: 0.2,
                                timeout: 18000,
                                modelOverride: fallbackModel,
                                requestScope: 'page'
                            });
                            if (gen !== siteGeneration) return;
                            it.translatedText = cleanSiteTrans(single);
                            it.status = 'done';
                            siteTranslatedCount = Array.from(siteItemsMap.values()).filter(x => x.status === 'done').length;
                            applyItemStyle(it);
                        } catch (se) {
                            if (gen !== siteGeneration) return;
                            it.status = 'failed';
                        }
                    }
                }
            } catch (err) {
                if (gen !== siteGeneration) return;
                for (const it of batch) {
                    try {
                        const single = await requestLLMChatCompletion({
                            messages: [
                                { role: "system", content: "Translate to Simplified Chinese. Return ONLY translated text." },
                                { role: "user", content: it.originalText }
                            ],
                            temperature: 0.2,
                            timeout: 18000,
                            modelOverride: fallbackModel,
                            requestScope: 'page'
                        });
                        if (gen !== siteGeneration) return;
                        it.translatedText = cleanSiteTrans(single);
                        it.status = 'done';
                        siteTranslatedCount = Array.from(siteItemsMap.values()).filter(x => x.status === 'done').length;
                        applyItemStyle(it);
                    } catch (se) {
                        if (gen !== siteGeneration) return;
                        it.status = 'failed';
                    }
                }
            }
        }

        function cleanSiteTrans(str) {
            if (!str) return '';
            let s = str.trim();
            if ((s.startsWith('"') && s.endsWith('"')) || (s.startsWith("'") && s.endsWith("'"))) {
                s = s.slice(1, -1).trim();
            }
            return s;
        }

            siteItemsMap.forEach(it => {
                applyItemStyle(it);
            });
            updateSiteFabState();
            showSiteToast('已切换为原生英文内容');
        }

        }

                    <button class="lexipro-site-menu-btn" id="lps-btn-auto-toggle">
                        <span>⚡ 进站自动翻译:</span>
                        <span id="lps-auto-status" style="margin-left:auto;font-weight:bold;color:${isCurrentSiteAutoTranslate() ? '#31c39a' : '#8b949f'};">${isCurrentSiteAutoTranslate() ? '当前网站开启' : '已关闭'}</span>
                    </button>
                </div>
                <div class="lexipro-site-fab-toolbar">
                    <div class="lexipro-site-fab" id="lexipro-site-fab">
                        <div class="lexipro-site-fab-icon">
                            <span class="lexipro-site-spinner" style="display:none;">⏳</span>
                            <span class="lexipro-site-logo-icon">MD</span>
                        </div>
                        <div class="lexipro-site-fab-label" id="lexipro-site-fab-label">${siteConfig.isGeneral ? '沉浸式翻译' : ('翻译' + siteConfig.name)}</div>
                    </div>
                    <button class="lexipro-site-fab-more" id="lexipro-site-fab-more" title="翻译选项与设置">⚙️</button>
                    <div class="lexipro-site-badge" id="lexipro-site-badge" style="display:none;">0</div>
                </div>
            `;

            document.body.appendChild(fabWrapper);

            const fab = fabWrapper.querySelector('#lexipro-site-fab');
            const fabMore = fabWrapper.querySelector('#lexipro-site-fab-more');
            const menu = fabWrapper.querySelector('#lexipro-site-menu');
            const btnTranslate = fabWrapper.querySelector('#lps-btn-translate');
            const btnRetry = fabWrapper.querySelector('#lps-btn-retry');
            const btnRestore = fabWrapper.querySelector('#lps-btn-restore');
            const btnNewItems = fabWrapper.querySelector('#lps-btn-new-items');

            updateStyleMenuUI();

            fab.addEventListener('click', (e) => {
                e.stopPropagation();
                if (!isSiteShowingTranslation && !isSiteTranslating) {
                    runSiteTranslation();
                } else if (isSiteShowingTranslation && !isSiteTranslating) {
                    // 已翻译状态下点击主按钮快速切换原文与中文
                    const isShowingOrig = Array.from(siteItemsMap.values()).some(it => it.isShowingOriginal);
                    siteItemsMap.forEach(it => {
                        it.isShowingOriginal = !isShowingOrig;
                        applyItemStyle(it);
                    });
                    updateSiteFabState();
                } else {
                    menu.classList.toggle('show');
                }
            });

            fabMore.addEventListener('click', (e) => {
                e.stopPropagation();
                menu.classList.toggle('show');
            });

            fab.addEventListener('contextmenu', (e) => {
                e.preventDefault();
                menu.classList.toggle('show');
            });

            btnTranslate.addEventListener('click', () => {
                menu.classList.remove('show');
                runSiteTranslation();
            });

            btnRetry.addEventListener('click', () => {
                menu.classList.remove('show');
                runSiteTranslation({ retryFailedOnly: true });
            });

            btnRestore.addEventListener('click', () => {
                menu.classList.remove('show');
                restoreSiteOriginal();
            });

            btnNewItems.addEventListener('click', () => {
                menu.classList.remove('show');
                runSiteTranslation();
            });

            const btnThemeToggle = fabWrapper.querySelector('#lps-btn-theme-toggle');
            const themeStatusText = fabWrapper.querySelector('#lps-theme-status');
            btnThemeToggle.addEventListener('click', () => {
                siteUITheme = siteUITheme === 'animal_island' ? 'market_desk' : 'animal_island';
                GM_setValue('lexipro_ui_theme', siteUITheme);
                themeStatusText.innerText = siteUITheme === 'animal_island' ? '动森经典' : 'Market Desk';
                const shadowHost = document.getElementById('imt-shadow-host');
                if (shadowHost) shadowHost.setAttribute('data-theme', siteUITheme);
                showSiteToast('已切换皮肤: ' + (siteUITheme === 'animal_island' ? '动森温馨经典' : 'Market Desk 金融终端高级质感'));
            });

            const btnAutoToggle = fabWrapper.querySelector('#lps-btn-auto-toggle');
            const autoStatusText = fabWrapper.querySelector('#lps-auto-status');
            function updateAutoStatusUI() {
                const isAuto = isCurrentSiteAutoTranslate();
                if (autoStatusText) {
                    autoStatusText.innerText = isAuto ? '当前网站开启' : '已关闭';
                    autoStatusText.style.color = isAuto ? '#31c39a' : '#8b949f';
                }
            }

            btnAutoToggle.addEventListener('click', () => {
                const host = (location && location.hostname ? location.hostname : '').toLowerCase();
                if (!host) return;
                let list = Array.isArray(CONFIG.autoTranslateSites) ? [...CONFIG.autoTranslateSites] : [];
                const exists = list.some(pat => matchesDomainPattern(host, pat));
                if (exists) {
                    list = list.filter(pat => !matchesDomainPattern(host, pat));
                } else {
                    list.push(host);
                }
                CONFIG.autoTranslateSites = list;
                GM_setValue('lexipro_auto_translate_sites', list);
                updateAutoStatusUI();
                showSiteToast('对此网站进站自动翻译: ' + (isCurrentSiteAutoTranslate() ? '已开启' : '已关闭'));
            });

            fabWrapper.querySelectorAll('.lexipro-style-option').forEach(opt => {
                opt.addEventListener('click', () => {
                    const chosenStyle = opt.dataset.style;
                    setTranslationStyle(chosenStyle);
                });
            });

            document.addEventListener('click', (e) => {
                if (!fabWrapper.contains(e.target)) {
                    menu.classList.remove('show');
                }
            });

            setupFabDrag(fabWrapper, fab);
        }

        function updateStyleMenuUI() {
            if (!fabWrapper) return;
            fabWrapper.querySelectorAll('.lexipro-style-option').forEach(opt => {
                const isActive = opt.dataset.style === siteTranslationStyle;
                opt.classList.toggle('active', isActive);
                const check = opt.querySelector('.check-icon');
                if (check) check.style.display = isActive ? 'inline' : 'none';
            });
        }

        function setupFabDrag(container, handle) {
            let dragging = false;
            let startX = 0, startY = 0;
            let initRight = 0, initBottom = 0;

            handle.addEventListener('mousedown', (e) => {
                if (e.button !== 0) return;
                dragging = false;
                startX = e.clientX;
                startY = e.clientY;

                const rect = container.getBoundingClientRect();
                initRight = window.innerWidth - rect.right;
                initBottom = window.innerHeight - rect.bottom;

                function onMove(ev) {
                    const dx = ev.clientX - startX;
                    const dy = ev.clientY - startY;
                    if (Math.abs(dx) > 4 || Math.abs(dy) > 4) {
                        dragging = true;
                        handle.classList.add('dragging');
                        const newR = Math.max(12, Math.min(window.innerWidth - 80, initRight - dx));
                        const newB = Math.max(12, Math.min(window.innerHeight - 80, initBottom - dy));
                        container.style.right = newR + 'px';
                        container.style.bottom = newB + 'px';
                    }
                }

                function onUp() {
                    document.removeEventListener('mousemove', onMove);
                    document.removeEventListener('mouseup', onUp);
                    handle.classList.remove('dragging');
                    if (dragging) {
                        const r = window.innerWidth - container.getBoundingClientRect().right;
                        const b = window.innerHeight - container.getBoundingClientRect().bottom;
                        GM_setValue('lexipro_site_fab_pos', { right: Math.round(r), bottom: Math.round(b) });
                    }
                }

                document.addEventListener('mousemove', onMove);
                document.addEventListener('mouseup', onUp);
            });
        }

        function updateSiteFabState() {
            if (!fabWrapper) return;
            const label = fabWrapper.querySelector('#lexipro-site-fab-label');
            const badge = fabWrapper.querySelector('#lexipro-site-badge');
            const spin = fabWrapper.querySelector('.lexipro-site-spinner');
            const logo = fabWrapper.querySelector('.lexipro-site-logo-icon');
            const btnTrans = fabWrapper.querySelector('#lps-btn-translate');
            const btnRetry = fabWrapper.querySelector('#lps-btn-retry');
            const btnRestore = fabWrapper.querySelector('#lps-btn-restore');
            const btnNew = fabWrapper.querySelector('#lps-btn-new-items');
            const statusText = fabWrapper.querySelector('#lexipro-site-status-text');

            const failedCount = Array.from(siteItemsMap.values()).filter(it => it.status === 'failed' || it.status === 'error').length;
            const isShowingOrig = Array.from(siteItemsMap.values()).some(it => it.isShowingOriginal);

            if (isSiteTranslating) {
                spin.style.display = 'inline-block';
                logo.style.display = 'none';
                label.innerText = `翻译中 ${siteTranslatedCount}/${siteTotalCount}`;
                statusText.innerText = '翻译中...';
            } else if (isSiteShowingTranslation) {
                spin.style.display = 'none';
                logo.style.display = 'inline-block';
                label.innerText = isShowingOrig ? `显示原文 (${siteTranslatedCount})` : `已翻译 (${siteTranslatedCount})`;
                btnTrans.style.display = 'none';
                btnRestore.style.display = 'flex';
                statusText.innerText = '已翻译';
            } else {
                spin.style.display = 'none';
                logo.style.display = 'inline-block';
                label.innerText = '翻译' + siteConfig.name;
                btnTrans.style.display = 'flex';
                btnRestore.style.display = 'none';
                statusText.innerText = '就绪';
            }

            if (btnRetry) {
                btnRetry.style.display = (!isSiteTranslating && failedCount > 0) ? 'flex' : 'none';
                if (failedCount > 0) {
                    btnRetry.innerText = `⚠️ 重试未完成段落 (${failedCount}条)`;
                }
            }

            if (siteUntranslatedNewCount > 0 && isSiteShowingTranslation) {
                badge.style.display = 'flex';
                badge.innerText = '+' + siteUntranslatedNewCount;
                btnNew.style.display = 'flex';
                btnNew.innerText = `✨ 翻译新增内容 (${siteUntranslatedNewCount}条)`;
            } else {
                badge.style.display = 'none';
                btnNew.style.display = 'none';
            }
        }

        function showSiteToast(msg) {
            let toast = document.getElementById('lexipro-site-toast');
            if (!toast) {
                toast = document.createElement('div');
                toast.id = 'lexipro-site-toast';
                toast.className = 'lexipro-site-toast';
                document.body.appendChild(toast);
            }
            toast.innerText = msg;
            toast.classList.add('show');
            setTimeout(() => { toast.classList.remove('show'); }, 3000);
        }

        let observerTimer = null;
        const siteObserver = new MutationObserver(() => {
            if (observerTimer) clearTimeout(observerTimer);
            observerTimer = setTimeout(() => {
                const found = collectSiteElements();
                if (found.length > 0 && isSiteShowingTranslation) {
                    siteUntranslatedNewCount += found.length;
                    updateSiteFabState();
                }
            }, 600);
        });

        // SPA 换页检测与生命周期隔离处理
        function onSiteUrlChange() {
            siteGeneration++;
            // 取消属于上一页的所有未完成请求
            activePageRequests.forEach(req => {
                try { req.abort(); } catch (e) {}
            });
            activePageRequests.clear();

            siteItemsMap.clear();
            siteItemCounter = 0;
            siteTranslatedCount = 0;
            siteTotalCount = 0;
            siteUntranslatedNewCount = 0;
            isSiteTranslating = false;
            isSiteShowingTranslation = false;

            setTimeout(() => {
                collectSiteElements();
                updateSiteFabState();
                if (siteAutoTranslate) {
                    runSiteTranslation();
                }
            }, 1000);
        }

        let lastSiteUrl = location.href;
        setInterval(() => {
            if (location.href !== lastSiteUrl) {
                lastSiteUrl = location.href;
                onSiteUrlChange();
            }
        }, 800);
        window.addEventListener('popstate', onSiteUrlChange);

        function destroySiteSpecificTranslator() {
            restoreSiteOriginal();
            if (siteObserver) {
                try { siteObserver.disconnect(); } catch (e) {}
            }
            if (fabWrapper && fabWrapper.parentNode) {
                fabWrapper.remove();
                fabWrapper = null;
            }
            siteItemsMap.clear();
            siteGeneration++;
            isSiteTranslating = false;
            isSiteShowingTranslation = false;
            window.__lexipro_site_initialized = false;
            if (typeof clearActiveWordHighlight === 'function') {
                clearActiveWordHighlight();
            }
        }

        // 注册给外层消息和生命周期调用的钩子
        window.__lexipro_site_initialized = true;
        window.__lexipro_destroy_site_translator = destroySiteSpecificTranslator;
        window.__lexipro_get_page_state = () => ({
            supported: true,
            connected: true,
            immersionActive: true,
            siteName: siteConfig.name,
            jobState: isSiteTranslating ? 'running' : (isSiteShowingTranslation ? 'completed' : 'idle'),
            totalCount: siteItemsMap.size,
            translatedCount: Array.from(siteItemsMap.values()).filter(x => x.status === 'done').length,
            failedCount: Array.from(siteItemsMap.values()).filter(x => x.status === 'failed' || x.status === 'error').length,
            pendingCount: Array.from(siteItemsMap.values()).filter(x => x.status === 'pending').length,
            isShowingOriginal: Array.from(siteItemsMap.values()).some(it => it.isShowingOriginal),
            displayMode: siteTranslationStyle,
            autoTranslate: isCurrentSiteAutoTranslate(),
            theme: siteUITheme
        });
        window.__lexipro_run_site_translation = (opts) => runSiteTranslation(opts);
        window.__lexipro_retry_failed = () => runSiteTranslation({ retryFailedOnly: true });
        window.__lexipro_toggle_original = () => {
            const isShowingOrig = Array.from(siteItemsMap.values()).some(it => it.isShowingOriginal);
            siteItemsMap.forEach(it => {
                it.isShowingOriginal = !isShowingOrig;
                applyItemStyle(it);
            });
            updateSiteFabState();
            return !isShowingOrig;
        };
        window.__lexipro_set_display_mode = (mode) => setTranslationStyle(mode);
        window.__lexipro_set_theme = (theme) => {
            siteUITheme = theme;
            GM_setValue('lexipro_ui_theme', theme);
            const themeStatusText = fabWrapper?.querySelector('#lps-theme-status');
            if (themeStatusText) themeStatusText.innerText = theme === 'animal_island' ? '动森经典' : 'Market Desk';
        };
        window.__lexipro_update_auto_status = () => {
            if (typeof updateAutoStatusUI === 'function') updateAutoStatusUI();
        };

        window.__lexipro_site_disconnect_observer = () => {
            if (siteObserver) {
                try { siteObserver.disconnect(); } catch (e) {}
            }
        };

        window.__lexipro_site_storage_listener = (key, value) => {
            if (key === 'lexipro_site_translation_style' && value) {
                siteTranslationStyle = value;
                updateStyleMenuUI();
                if (isSiteShowingTranslation) {
                    siteItemsMap.forEach(it => { if (it.status === 'done') applyItemStyle(it); });
                }
            } else if (key === 'lexipro_auto_translate_sites') {
                if (typeof updateAutoStatusUI === 'function') updateAutoStatusUI();
            } else if (key === 'lexipro_ui_theme' && value) {
                siteUITheme = value;
                const themeStatusText = fabWrapper?.querySelector('#lps-theme-status');
                if (themeStatusText) themeStatusText.innerText = value === 'animal_island' ? '动森经典' : 'Market Desk';
            }
        };

        function startSiteEngine() {
            window.__lexipro_site_initialized = true;
            buildSiteFab();
            collectSiteElements();
            if (document.body) {
                siteObserver.observe(document.body, { childList: true, subtree: true });
            }

            // 进站自动翻译特性支持：严格限制在白名单中，默认绝不自动翻译！
            if (isCurrentSiteAutoTranslate()) {
                setTimeout(() => {
                    runSiteTranslation();
                }, 1500);
            }
        }

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', startSiteEngine);
        } else {
            startSiteEngine();
        }
    }

    // 扩展环境与 Popup 的全局消息交互支持
    if (typeof chrome !== 'undefined' && chrome.runtime && chrome.runtime.onMessage) {
        chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
            if (msg.type === 'GET_PAGE_STATE') {
                if (window.__lexipro_site_initialized && typeof window.__lexipro_get_page_state === 'function') {
                    sendResponse(window.__lexipro_get_page_state());
                } else {
                    sendResponse({
                        supported: true,
                        connected: true,
                        immersionActive: false,
                        siteName: '通用网页',
                        jobState: 'idle',
                        totalCount: 0,
                        translatedCount: 0,
                        failedCount: 0,
                        pendingCount: 0,
                        isShowingOriginal: false,
                        displayMode: GM_getValue('lexipro_site_translation_style', 'replace'),
                        autoTranslate: isCurrentSiteAutoTranslate(),
                        theme: GM_getValue('lexipro_ui_theme', 'market_desk')
                    });
                }
                return true;
            }
            if (msg.type === 'ENABLE_IMMERSION_FOR_SITE') {
                CONFIG.isSessionImmersionEnabled = true;
                if (!window.__lexipro_site_initialized) {
                    initSiteSpecificTranslator();
                }
                sendResponse({ success: true, enabled: true });
                return true;
            }
            if (msg.type === 'DISABLE_IMMERSION_FOR_SITE') {
                CONFIG.isSessionImmersionEnabled = false;
                if (typeof window.__lexipro_destroy_site_translator === 'function') {
                    window.__lexipro_destroy_site_translator();
                }
                sendResponse({ success: true, disabled: true });
                return true;
            }
            if (msg.type === 'START_PAGE_TRANSLATION') {
                CONFIG.isSessionImmersionEnabled = true;
                if (!window.__lexipro_site_initialized) {
                    initSiteSpecificTranslator();
                }
                if (typeof window.__lexipro_run_site_translation === 'function') {
                    window.__lexipro_run_site_translation();
                }
                sendResponse({ success: true });
                return true;
            }
            if (msg.type === 'RETRY_FAILED') {
                if (typeof window.__lexipro_retry_failed === 'function') {
                    window.__lexipro_retry_failed();
                }
                sendResponse({ success: true });
                return true;
            }
            if (msg.type === 'TOGGLE_ORIGINAL') {
                if (typeof window.__lexipro_toggle_original === 'function') {
                    const isOrig = window.__lexipro_toggle_original();
                    sendResponse({ success: true, isShowingOriginal: isOrig });
                } else {
                    sendResponse({ success: false });
                }
                return true;
            }
            if (msg.type === 'SET_DISPLAY_MODE') {
                if (typeof window.__lexipro_set_display_mode === 'function') {
                    window.__lexipro_set_display_mode(msg.mode);
                } else {
                    GM_setValue('lexipro_site_translation_style', msg.mode);
                }
                sendResponse({ success: true });
                return true;
            }
            if (msg.type === 'SET_THEME') {
                if (typeof window.__lexipro_set_theme === 'function') {
                    window.__lexipro_set_theme(msg.theme);
                } else {
                    GM_setValue('lexipro_ui_theme', msg.theme);
                }
                sendResponse({ success: true });
                return true;
            }
        });
    }

    // 全局配置热更新监听（由 storage 监听回调派发）
    if (typeof window !== 'undefined') {
        window.__lexipro_onStorageChange = (key, value) => {
            if (key === 'lexipro_excluded_sites') {
                handleExcludedSitesChange(value);
            } else if (key === 'lexipro_auto_translate_sites') {
                CONFIG.autoTranslateSites = Array.isArray(value) ? value : [];
                if (typeof window.__lexipro_update_auto_status === 'function') {
                    window.__lexipro_update_auto_status();
                }
            } else if (key === 'lexipro_immersion_sites') {
                CONFIG.immersionSites = Array.isArray(value) ? value : [];
                const shouldEnable = isCurrentSiteImmersionEnabled();
                if (shouldEnable && !window.__lexipro_site_initialized) {
                    initSiteSpecificTranslator();
                } else if (!shouldEnable && window.__lexipro_site_initialized) {
                    if (typeof window.__lexipro_destroy_site_translator === 'function') {
                        window.__lexipro_destroy_site_translator();
                    }
                }
            } else if (key === 'lexipro_auto_pronounce') {
                CONFIG.autoPronounce = Boolean(value);
            } else if (key === 'lexipro_tts_engine') {
                CONFIG.ttsEngine = value || 'youdao';
            } else if (key === 'lexipro_kokoro_voice') {
                CONFIG.kokoroVoice = value || 'af_heart';
            } else if (key === 'lexipro_local_tts_url') {
                CONFIG.localTtsUrl = value || 'http://127.0.0.1:18080';
            } else if (key === 'lexipro_disable_thinking') {
                CONFIG.disableThinking = Boolean(value);
                if (CONFIG.llmProviders?.siliconflow) CONFIG.llmProviders.siliconflow.disableThinking = CONFIG.disableThinking;
                if (CONFIG.llmProviders?.kimi_free) CONFIG.llmProviders.kimi_free.enableThinking = !CONFIG.disableThinking;
            } else if (key === 'lexipro_notion_token') {
                CONFIG.notionToken = value || '';
                if (!isCurrentSiteExcluded() && CONFIG.notionToken && CONFIG.notionDatabaseId) {
                    syncVocabData();
                }
            } else if (key === 'lexipro_notion_database_id') {
                let cleanDb = value || '';
                const m = cleanDb.match(/([a-f0-9]{32})/i) || cleanDb.match(/([a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12})/i);
                if (m) cleanDb = m[1].replace(/-/g, '');
                CONFIG.notionDatabaseId = cleanDb;
                if (!isCurrentSiteExcluded() && CONFIG.notionToken && CONFIG.notionDatabaseId) {
                    syncVocabData();
                }
            }
            if (typeof window.__lexipro_site_storage_listener === 'function') {
                window.__lexipro_site_storage_listener(key, value);
            }
        };
    }

    // 启动系统核心初始化入口
    init();

})();
